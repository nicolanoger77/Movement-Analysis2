2. MOVEMENT FEATURE ENGINEERING

## Prepare numeric coordinates for efficient interpolation
coords <- st_coordinates(tracks)
tracks$X <- coords[,1]
tracks$Y <- coords[,2]

# Calculate baseline movement parameters based on spatial coordinates and time
tracks_features <- tracks |>
  arrange(user_id, datetime) |>
  group_by(user_id) |>
  mutate(
    lead_X = lead(X),
    lead_Y = lead(Y),
    time_diff = as.numeric(difftime(lead(datetime), datetime, units = "secs")),
    # Euclidean distance in projected CRS (fast) as segment length
    distance = sqrt((lead_X - X)^2 + (lead_Y - Y)^2),
    speed = ifelse(time_diff > 0, distance / time_diff, NA_real_),
    acceleration = (lead(speed) - speed) / time_diff,
    # mark segments where a valid next point exists
    has_next = !is.na(lead_X) & !is.na(lead_Y) & time_diff > 0
  ) |>
  ungroup()

# Keep only valid numeric movement observations
tracks_features <- tracks_features |>
  filter(!is.na(speed) & is.finite(speed))

3. SPATIAL CONTEXT INTEGRATION

## Simplify context integration by directly using the original GPS points.
## We take a clean sub-sample of GPS points for performance, but keep 
## enough data to build a solid model.

set.seed(42)
# Keep up to 10,000 points to keep distance calculations fast and modelling robust
tracks_context <- tracks_features |>
  slice_sample(n = min(100000, nrow(tracks_features)))

# Compute nearest highway for each sampled point
nearest_hw_idx <- st_nearest_feature(tracks_context, highway)
tracks_context$distance_to_highway <- as.numeric(st_distance(tracks_context, highway[nearest_hw_idx, ], by_element = TRUE))
tracks_context$highway_type <- highway$highway[nearest_hw_idx]
tracks_context$cycleway <- highway$cycleway[nearest_hw_idx]

# Compute nearest railway for each sampled point
nearest_rail_idx <- st_nearest_feature(tracks_context, railway)
tracks_context$distance_to_railway <- as.numeric(st_distance(tracks_context, railway[nearest_rail_idx, ], by_element = TRUE))

4. CUSTOM FEATURE ENGINEERING

# Feature 2: Binary indicator for immediate proximity to railway tracks (less than 50 meters)
tracks_context$close_to_railway <- tracks_context$distance_to_railway < 50

# Feature 3: Binary indicator separating major roads
major_road_types <- c("motorway", "trunk", "primary", "secondary")
tracks_context$is_major_road <- tracks_context$highway_type %in% major_road_types

5. DATA CLEANING & PREPARATION FOR ML

# Handle missing values and convert categorical strings into factors

# Define feature sets for comparison
movement_features <- c("speed", "acceleration")
context_features <- c("speed", "acceleration", "highway", "cycleway", 
                      "distance_to_railway", "distance_to_highway", 
                      "close_to_railway", "is_major_road")

# Create data split for training (80%) and testing (20%)
set.seed(42)
train_indices <- createDataPartition(ml_ready_data$mode, p = 0.8, list = FALSE)
train_set <- ml_ready_data[train_indices, ]
test_set <- ml_ready_data[-train_indices, ]

# Define lightweight training parameters
formula_m1 <- as.formula(paste("mode ~", paste(movement_features, collapse = " + ")))
formula_m2 <- as.formula(paste("mode ~", paste(context_features, collapse = " + ")))

# Train models using ranger (Random Forest)
